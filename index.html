// 修改後的 saveEdit function saveEdit(id) { const elItem = document.getElementById(`editItem_${id}`); const elPrice = document.getElementById(`editPrice_${id}`); const elPay = document.getElementById(`editPay_${id}`); if (!elItem || !elPrice || !elPay) { alert("找不到編輯欄位，請重新整理！"); return; } const newItem = elItem.value.trim(); const newPrice = parseInt(elPrice.value) || 0; const newPay = parseInt(elPay.value) || 0; if (!newItem) { alert("品項不能是空白的喔！"); return; } // 關鍵修改：先重置狀態，再發送更新 editingId = null; database.ref('orders').child(id).update({ item: newItem, price: newPrice, pay: newPay }).catch(err => { alert("儲存失敗：" + err.message); // 如果失敗，則恢復編輯狀態 editingId = id; renderTable(currentOrdersData); // 這裡需要能取得當前資料 }); // 不要在這裡呼叫 renderTable，Firebase 的 .on('value') 會自動幫你做 }
